\textbf{Geometric Origin}: Ratio of Y-junction (proton) to spherical defect (electron) energies.

\begin{equation}
\frac{E_p}{E_e} = 6\pi^5 = 1836.1181...

\begin{equation}
\frac{m_p}{m_e} \equiv \frac{E_p}{E_e} \quad\text{(mass = rest energy)}
\label{eq:mpme_bridge}

\subsubsection{Neutron-Proton Mass Difference}

\textbf{Geometric Origin}: Asymmetric Y-junction ($60^\circ$ angle) vs symmetric ($120^\circ$).

\textbf{Model result} \tagDerSym{}:
\begin{equation}
\Delta E_{np} = E_n - E_p = \frac{8 E_e}{\pi}
\label{eq:deltaEnp_bridge}
\end{equation}

\textbf{Dictionary step} \tagDc{}:
\begin{equation}
\Delta m_{np} \equiv \Delta E_{np}/c^2, \quad E_e \equiv m_e c^2
\quad\Rightarrow\quad
\Delta m_{np} = \frac{8 m_e}{\pi} = 1.301\,\text{MeV}
\label{eq:deltamnp_bridge}
\end{equation}

\textbf{Baseline} \tagBL{}: $\Delta m_{np} = 1.29333236 \pm 0.00000046$ MeV (\refBaseline)

\textbf{Agreement}: $0.6\%$

\textbf{Part I Location}: Chapter 4 (Confinement and the Strong Force)
