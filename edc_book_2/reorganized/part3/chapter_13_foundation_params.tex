\section{Introduction: The Parameter Problem}
\label{sec:foundation_intro}
% ════════════════════════════════════════════════════════════════════════════════

The Standard Model has 19+ free parameters that are measured, not derived. EDC aims
to reduce this by deriving parameters from 5D membrane geometry. This chapter addresses
three foundational Open Problem Reports (OPRs):

\begin{itemize}[nosep]
\item \textbf{OPR-01}: The membrane tension $\sigma$ and its relation to the mass scale $M_0$
\item \textbf{OPR-04}: The brane thickness $\Delta$ from scalar kink theory
\item \textbf{OPR-19}: The 5D gauge coupling $g_5$ and its reduction to 4D
\end{itemize}

\begin{statusbox}{Epistemic Framework}
\textbf{What this chapter achieves:}
\begin{itemize}[nosep]
\item Derives \emph{relationships} between parameters (reducing freedom) \tagDc{}
\item Establishes the Scale Taxonomy (canonical reference for all derivations)
\item Provides no-smuggling certification for each derivation
\end{itemize}

\textbf{What remains open:}
\begin{itemize}[nosep]
\item Absolute values of primitive parameters ($\sigma$, $v$, $y$) remain \tagP{}
\item Scale identifications (A1)--(A3) are assumptions, not derivations
\end{itemize}
\end{statusbox}

\section{OPR-01: Membrane Tension and Mass Scale}
\label{sec:opr01}
% ════════════════════════════════════════════════════════════════════════════════

\begin{tcolorbox}[edcGuardrail, title=\textbf{Epistemic Status: OPR-01}]
This section derives the relationship between the bulk mass scale $M_0$ and the
membrane tension $\sigma$:
\begin{itemize}[nosep]
    \item $M_0 = f(\sigma, \Delta, y)$ derived from domain-wall physics \tagDc{}
    \item Domain-wall ansatz for $M(\xi)$ remains \tagP{}
    \item Yukawa coupling $y$ remains \tagP{}
\end{itemize}
\textbf{What is NOT claimed:} We do not derive $\sigma$, $\Delta$, or $y$ from first
principles. This derivation shows \emph{how} these parameters are related, converting
three independent \tagP{} parameters into two independent + one derived \tagDc{}.
\end{tcolorbox}

\subsection{The Domain-Wall Ansatz}

The thick brane in EDC is modeled as a domain wall generated by a scalar field
$\phi(\xi)$ with a double-well potential \tagP{}:
\begin{equation}
    V(\phi) = \frac{\lambda}{4} \left(\phi^2 - v^2\right)^2
    \label{eq:opr01_potential}
\end{equation}

The static kink solution interpolating between the two vacua $\phi = \pm v$
is a standard consequence of the Euler--Lagrange equation for
Eq.~(\ref{eq:opr01_potential}) \tagDer{}:
\begin{equation}
    \phi(\xi) = v \tanh\left(\frac{\xi}{\Delta}\right)
    \label{eq:opr01_kink}
\end{equation}
where the wall thickness is:
\begin{equation}
    \Delta = \frac{2}{v\sqrt{\lambda}}
    \label{eq:opr01_Delta}
\end{equation}

The bulk fermion acquires a position-dependent mass through Yukawa coupling \tagP{}:
\begin{equation}
    M(\xi) = y \, \phi(\xi) = M_0 \tanh\left(\frac{\xi}{\Delta}\right)
    \quad \text{with} \quad \boxed{M_0 = y v}
    \label{eq:opr01_M0}
\end{equation}

\subsection{Domain-Wall Tension}

The tension $\sigma$ is the integrated energy density per unit transverse area.
Assuming a BPS-saturated (minimum-tension) wall within this effective scalar model \tagP{}:
\begin{equation}
    \sigma = \int_{-\infty}^{\infty} T_{00}(\xi) \, d\xi = \frac{2\lambda v^4 \Delta}{3}
\end{equation}

Substituting $\lambda = 2/(\Delta^2 v^2)$:
\begin{equation}
    \boxed{\sigma \Delta = \frac{4 v^2}{3}}
    \label{eq:opr01_sigmaDelta}
\end{equation}

This is the \textbf{BPS constraint} connecting tension, thickness, and scalar VEV.

\begin{tcolorbox}[colback=yellow!5!white, colframe=yellow!50!black,
    title=\textbf{Dictionary Note (Scope)}]
The relations in this subsection are \emph{exact within the scalar kink model}.
Identifying $\sigma_{\text{kink}}$ with the EDC membrane tension $\sigma_{\text{EDC}}$
is a conditional dictionary assumption \tagDc{} (not a pure theorem).
\end{tcolorbox}

\subsection{Main Result: $M_0$ from $\sigma$ and $\Delta$}

Combining $v = M_0/y$ with the BPS constraint:
\begin{equation}
    \boxed{M_0 = \frac{\sqrt{3}}{2} \, y \, \sqrt{\sigma \Delta}}
    \quad \tagDcSym{}
    \label{eq:opr01_result}
\end{equation}

\textbf{Parameter status:} The Yukawa factor $y$ is not yet derived from geometry
and is treated as \tagOpen{} (or \tagCal{} if later anchored). Therefore $M_0$ is a
conditional scale \tagDc{}, not a prediction.

\textbf{Consequence for three generations (OPR-21):}
The dimensionless parameter $\mu = M_0 \ell$ controls bound state count. If $\ell = n\Delta$
(assumption A3) \tagP{}:
\begin{equation}
    \mu = \frac{\sqrt{3}}{2} \, y \, n \, \sqrt{\sigma \Delta^3}
    \label{eq:opr01_mu}
\end{equation}

\section{OPR-04: Wall Thickness from Kink Theory}
\label{sec:opr04}
% ════════════════════════════════════════════════════════════════════════════════

\begin{tcolorbox}[edcGuardrail, title=\textbf{Epistemic Status: OPR-04}]
\textbf{Goal:} Derive the domain-wall thickness $\Delta$ from the scalar kink
solution of $\lambda\phi^4$ theory.

\textbf{Key result:} $\Delta = 2/(v\sqrt{\lambda})$ where $v$ is the vacuum
expectation value and $\lambda$ is the self-coupling.

\textbf{Status:} \tagDc{} (conditional on $v$, $\lambda$ remaining \tagP{})
\end{tcolorbox}

\subsection{Derivation from BPS Equation}

The static kink satisfies the BPS equation (first-order reduction of EOM) \tagDer{}:
\begin{equation}
    \frac{d\phi}{d\xi} = \pm \sqrt{2 V(\phi)}
    \label{eq:opr04_BPS}
\end{equation}

Substituting the tanh ansatz and equating coefficients:
\begin{equation}
    \frac{v}{\Delta} = v^2 \sqrt{\frac{\lambda}{2}}
    \quad \Rightarrow \quad
    \boxed{\Delta = \frac{2}{v\sqrt{\lambda}} = \sqrt{\frac{2}{\lambda}} \cdot \frac{1}{v}}
    \quad \tagDer{}
    \label{eq:opr04_result}
\end{equation}

\textbf{Physical interpretation:}
\begin{itemize}[nosep]
    \item Large $v$ $\to$ thin wall (higher VEV compresses the transition)
    \item Large $\lambda$ $\to$ thin wall (stronger self-coupling steepens the potential)
    \item The combination $v\sqrt{\lambda}$ sets the inverse length scale
\end{itemize}

\subsection{Parameter Reduction}

The BPS relation $\sigma\Delta = 4v^2/3$ \textbf{eliminates $\lambda$} as an
independent parameter. Once $\sigma$ and $v$ are specified, $\Delta$ is determined:
\begin{equation}
    \Delta = \frac{4 v^2}{3 \sigma}
    \quad \tagDc{}
    \label{eq:opr04_Delta_from_sigma}
\end{equation}

\section{OPR-19: 5D Gauge Coupling Reduction}
\label{sec:opr19}
% ════════════════════════════════════════════════════════════════════════════════

\begin{tcolorbox}[edcGuardrail, title=\textbf{Epistemic Status: OPR-19}]
\textbf{Goal:} Derive the canonical relation between the 5D gauge coupling $g_5$
and the effective 4D coupling $g_4$ by explicit dimensional reduction.

\textbf{Inputs:}
\begin{itemize}[nosep]
    \item A stated 5D metric ansatz (warped geometry) \tagP{}
    \item The 5D gauge kinetic term with proper measure $\sqrt{-G}$
    \item No SM observables used as anchors
\end{itemize}

\textbf{Status:} CONDITIONAL \tagDc{} --- formula derived; warp factor and domain remain \tagP{}.
\end{tcolorbox}

\subsection{The 5D Gauge Action}

Starting from the 5D gauge field action with canonical normalization:
\begin{equation}
    S_{\text{gauge}}^{(5D)} = -\frac{1}{4g_5^2} \int d^5x \sqrt{-G} \,
    G^{MA} G^{NB} F_{MN} F_{AB}
    \label{eq:opr19_action}
\end{equation}

With the warped metric ansatz \tagP{}:
\begin{equation}
    ds^2 = e^{2A(\xi)} \eta_{\mu\nu} dx^\mu dx^\nu + d\xi^2
    \label{eq:opr19_metric}
\end{equation}

\subsection{Critical Result: Warp Factor Cancellation}

Decomposing into KK modes $A_\mu(x,\xi) = \sum_n a_\mu^{(n)}(x) f_n(\xi)$ and
computing the 4D effective action:

\textbf{Scope assumptions (for the cancellation below):}
\begin{itemize}[nosep]
\item $A_5 = 0$ gauge; we keep only the $F_{\mu\nu}$ kinetic term.
\item Zero-mode reduction with separable form $A_\mu(x,\xi)=a_\mu(x)f(\xi)$.
\item Mode normalization is specified explicitly below (this choice affects the $g_5 \to g_4$ mapping).
\end{itemize}

\begin{tcolorbox}[colback=green!5!white, colframe=green!50!black,
    title=\textbf{Warp Factor Cancellation}]
\begin{equation}
    \sqrt{-G} \cdot F_{\mu\nu} F^{\mu\nu} = f_{\mu\nu}^{(n)} f^{(n)\mu\nu} |f_n(\xi)|^2
    \label{eq:opr19_cancellation}
\end{equation}

\textbf{The warp factors $e^{4A}$ from $\sqrt{-G}$ and $e^{-4A}$ from the
index contractions exactly cancel!}

The 4D gauge kinetic term has a \emph{flat measure} in the $\xi$-integral,
regardless of the warp factor profile.
\end{tcolorbox}

\subsection{Main Result: $g_5 \to g_4$ Reduction}

\begin{tcolorbox}[colback=red!5!white, colframe=red!50!black,
    title=\textbf{Result: Coupling Reduction Formula}]
\begin{equation}
    \boxed{\frac{1}{g_{4,n}^2} = \frac{1}{g_5^2} \int_0^\ell d\xi \, |f_n(\xi)|^2}
    \quad \tagDcSym{}
    \label{eq:opr19_result}
\end{equation}

\textbf{Weight function:} $W(\xi) = 1$ (flat measure, due to warp cancellation).

\textbf{Special case (flat zero-mode):}
For $f_0 = 1/\sqrt{\ell}$ one has $\int_0^\ell d\xi\,|f_0|^2 = 1$, so
\begin{equation}
    \frac{1}{g_4^2} = \frac{1}{g_5^2} \cdot 1
    \label{eq:opr19_flat_integral}
\end{equation}

\textbf{Dimensional convention:} In standard 5D normalization, $g_5$ has dimension
$[\text{length}]^{1/2}$ (or $[\text{mass}]^{-1/2}$ in natural units).
The \emph{dimensionless} 4D coupling is:
\begin{equation}
    \boxed{g_4 = \frac{g_5}{\sqrt{\ell}}} \quad \tagDer{}
    \label{eq:opr19_flat}
\end{equation}

\textbf{Alternative convention:} Defining $\tilde{g}_5 \equiv g_5/\sqrt{\ell}$
(dimensionless), one may write $g_4 = \tilde{g}_5$. This convention is sometimes
used implicitly; we use the standard (dimensionful $g_5$) convention throughout.
\end{tcolorbox}
