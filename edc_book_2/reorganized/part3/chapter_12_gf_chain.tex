\section{Introduction: The Fermi Puzzle}
\label{sec:gf_intro}
% ════════════════════════════════════════════════════════════

The Fermi coupling constant is one of the most precisely measured
quantities in physics:
\begin{equation}
G_F = 1.1663787(6) \times 10^{-5} \text{ GeV}^{-2} \quad \tagBL
\end{equation}

\textbf{The puzzle}: Why this value? The Standard Model treats $G_F$ as
an input parameter, derived from:
\begin{equation}
G_F = \frac{\pi \alpha}{\sqrt{2} M_W^2 \sin^2\theta_W} \quad \text{(SM relation)}
\end{equation}

But this just moves the question to $M_W$ and $\sin^2\theta_W$---still
unexplained.

\textbf{EDC approach}: $G_F$ emerges from a chain of reductions:
\begin{equation}
g_5 \xrightarrow{\text{compactification}} g_4 \xrightarrow{\text{localization}}
G_{\text{eff}} \xrightarrow{\text{normalization}} G_F
\end{equation}

\section{5D Coupling $g_5$}
\label{sec:g5}
% ════════════════════════════════════════════════════════════

\subsection{Definition}

In 5D, the fundamental coupling has dimensions:
\begin{equation}
[g_5] = \text{Length}^{1/2} \quad \text{(in natural units)} \quad \tagM
\end{equation}

\textbf{Physical interpretation} \tagP{}: $g_5$ characterizes the interaction
strength between 5D fields at the membrane level.

\subsection{Geometric Origin}

\begin{mechanism}{$g_5$ from Membrane Geometry}
\textbf{Ansatz} \tagP{}:
The 5D coupling is related to membrane tension and thickness:
\begin{equation}
g_5 \sim \sqrt{\frac{\hbar c}{\sigma}} \cdot \delta^{-1/2} \quad \tagP
\end{equation}

where:
\begin{itemize}[nosep]
\item $\sigma$: Membrane tension (MeV/fm$^2$)
\item $\delta$: Brane thickness (fm)
\end{itemize}

\textbf{Status}: This is a postulate [P]. Deriving $g_5$ from first
principles is \OPR{19}.
\end{mechanism}

\subsection{Dimensional Analysis}

From the membrane parameters (Chapter~\ref{ch:foundation_params}):
\begin{align}
\sigma &\approx 8.82 \text{ MeV/fm}^2 \quad \tagDc \\
\delta &\sim 10^{-3} \text{ fm} \quad \tagP
\end{align}

This gives:
\begin{equation}
g_5 \sim 0.1 \text{ fm}^{1/2} \quad \tagDcApprox
\end{equation}

% ════════════════════════════════════════════════════════════
\section{Reduction to 4D: $g_5 \to g_4$}
\label{sec:reduction}
% ════════════════════════════════════════════════════════════

\subsection{Compactification}

If the 5th dimension has size $R_\xi$, dimensional reduction gives:
\begin{equation}
g_4 = \frac{g_5}{\sqrt{R_\xi}} \quad \tagDer
\end{equation}

This is a standard result from Kaluza-Klein theory.

\subsection{The Role of $R_\xi$}

\begin{mechanism}{Compactification Scale}
\textbf{Definition} \tagP{}:
$R_\xi$ is the characteristic size of the compact 5th dimension.

\textbf{Scale hierarchy}:
\begin{equation}
R_\xi \ll r_p \quad \text{(sub-proton scale)} \quad \tagP
\end{equation}

\textbf{Constraint from $M_W$}:
\begin{equation}
M_W \sim \frac{\hbar c}{R_\xi} \approx 80 \text{ GeV} \quad \implies \quad
R_\xi \sim 2.5 \times 10^{-3} \text{ fm} \quad \tagI
\end{equation}
\end{mechanism}

\subsection{Effective 4D Coupling}

\begin{equation}
g_4 = \frac{g_5}{\sqrt{R_\xi}} \sim 0.1 \cdot (R_\xi)^{-1/2} \sim 0.6 \quad \tagDcApprox
\end{equation}

This is order-unity, consistent with electroweak coupling strength.

\section{Localization: $g_4 \to G_{\text{eff}}$}
\label{sec:localization}
% ════════════════════════════════════════════════════════════

\subsection{Brane Localization Suppression}

Not all fields are equally localized on the brane. The effective coupling
depends on wavefunction overlap:

\begin{equation}
G_{\text{eff}} = g_4^2 \cdot \int dy \, |\psi_L(y)|^2 |\psi_R(y)|^2 \quad \tagDer
\end{equation}

where $\psi_{L,R}(y)$ are fermion wavefunctions in the 5th dimension.

\subsection{Chiral Suppression}

\begin{mechanism}{V$-$A from Localization}
\textbf{Key insight} \tagDc{}:
If left-handed and right-handed fermions are localized on
\textit{opposite sides} of the brane:

\begin{center}
\begin{tikzpicture}[scale=0.8]

\node at (0,0) {Brane};

\node[blue] at (-2,2) {$\psi_L$};
\draw[thick, red] plot[smooth] coordinates {(-3,0.1) (-2,0.3) (-1,0.8) (0,1.5) (1,2) (2,1.5) (3,0.8)};
\node[red] at (2,2) {$\psi_R$};
\end{tikzpicture}
\end{center}

Then the overlap integral is suppressed:
\begin{equation}
\int |\psi_L|^2 |\psi_R|^2 \, dy \ll 1 \quad \tagDc
\end{equation}

This geometrically produces V$-$A structure (Chapter~\ref{ch:va_structure}).
\end{mechanism}

\section{Fermi Coupling $G_F$}
\label{sec:gf_derivation}
% ════════════════════════════════════════════════════════════

\subsection{From Effective Coupling to $G_F$}

The Fermi constant is defined via the muon decay rate:
\begin{equation}
\frac{1}{\tau_\mu} = \frac{G_F^2 m_\mu^5}{192 \pi^3} \quad \text{(tree level)}
\end{equation}

In EDC:
\begin{equation}
G_F = C_{\text{red}}^{(G_F)} \cdot \frac{g_4^2}{8 M_W^2} \quad \tagDc
\end{equation}

where $C_{\text{red}}^{(G_F)}$ is the reduction normalization factor.

\subsection{Consistency Check}

\begin{mechanism}{$G_F$ Consistency}
\textbf{Inputs} \tagBL{}:
\begin{align}
M_W &= 80.377 \text{ GeV} \\
\sin^2\theta_W &= 0.23121 \\
\alpha &= 1/137.036
\end{align}

\textbf{SM relation}:
\begin{equation}
G_F = \frac{\pi \alpha}{\sqrt{2} M_W^2 \sin^2\theta_W} = 1.166 \times 10^{-5} \text{ GeV}^{-2}
\end{equation}

\textbf{EDC check}: Using geometric values of $M_W$ (from $R_\xi$) and
$\sin^2\theta_W = 1/4$ (from $Z_6$), the EDC chain reproduces $G_F$
to within the expected uncertainty from $C_{\text{red}}$ [Open].
\end{mechanism}
