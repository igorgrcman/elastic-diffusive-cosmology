\section{Introduction: Why Boundary Value Problems?}
\label{sec:bvp_intro}
% ════════════════════════════════════════════════════════════

In EDC, particle properties emerge from the geometry of 5D structures.
Mathematically, this means solving equations of the form:
\begin{equation}
\mathcal{L}[\psi] = 0 \quad \text{with boundary conditions at } y = \pm \delta/2
\end{equation}

where $\mathcal{L}$ is a differential operator and $y$ is the coordinate
transverse to the brane.

\textbf{Key applications}:
\begin{itemize}[nosep]
\item Fermion wavefunctions $\to$ mass eigenvalues
\item Boson localization $\to$ $M_W$, $M_Z$
\item Tunneling rates $\to$ decay lifetimes
\item Mode spectrum $\to$ generation structure
\end{itemize}

\section{General BVP Setup}
\label{sec:bvp_setup}
% ════════════════════════════════════════════════════════════

\subsection{The Domain}

The thick brane occupies a slab in the 5th dimension:
\begin{equation}
y \in \left[-\frac{\delta}{2}, +\frac{\delta}{2}\right]
\end{equation}

with the brane centered at $y = 0$.

\subsection{Generic Equation}

For a field $\psi(y)$ confined to the brane:
\begin{equation}
-\frac{d^2\psi}{dy^2} + V(y)\psi = \lambda \psi \quad \tagDer
\label{eq:bvp_generic}
\end{equation}

where:
\begin{itemize}[nosep]
\item $V(y)$: Effective potential from brane profile
\item $\lambda$: Eigenvalue (related to 4D mass)
\end{itemize}

\subsection{Boundary Conditions}

\begin{mechanism}{Boundary Conditions}
\textbf{Types} \tagP{}:
\begin{enumerate}[nosep]
\item \textbf{Dirichlet}: $\psi(\pm\delta/2) = 0$ (vanishing at boundary)
\item \textbf{Neumann}: $\psi'(\pm\delta/2) = 0$ (vanishing derivative)
\item \textbf{Robin}: $\psi' + \kappa\psi = 0$ at boundary (mixed)
\item \textbf{Periodic}: $\psi(\delta/2) = \psi(-\delta/2)$
\end{enumerate}

\textbf{Physical meaning}:
\begin{itemize}[nosep]
\item Dirichlet: Field cannot penetrate into bulk
\item Neumann: Field extends smoothly into bulk
\item Robin: Partial reflection at brane-bulk interface
\end{itemize}

\textbf{Default in EDC}: Robin BC with parameter $\kappa$ determined by
membrane physics.
\end{mechanism}

% ════════════════════════════════════════════════════════════
\section{Robin Boundary Conditions}
\label{sec:robin_bc}
% ════════════════════════════════════════════════════════════

\subsection{Mathematical Form}

The Robin condition at the boundaries:
\begin{equation}
\psi' + \kappa_{\pm} \psi = 0 \quad \text{at } y = \pm\delta/2 \quad \tagP
\end{equation}

where $\kappa_{\pm}$ may differ for the two boundaries
(asymmetric brane).

\subsection{Eigenvalue Equation}

For a square-well potential $V(y) = 0$ inside the brane, the eigenvalue
condition becomes \tagDer{}:
\begin{equation}
\tan(k\delta) = \frac{k(\kappa_+ + \kappa_-)}{k^2 - \kappa_+\kappa_-}
\label{eq:robin_eigenvalue}
\end{equation}

where $k = \sqrt{\lambda}$ and $\lambda$ is the eigenvalue.

\subsection{Limiting Cases}

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Limit} & \textbf{Condition} & \textbf{Spectrum} \\
\midrule
$\kappa \to \infty$ & Dirichlet & $k_n = n\pi/\delta$ \\
$\kappa \to 0$ & Neumann & $k_n = n\pi/\delta$ (shifted) \\
Finite $\kappa$ & Robin & Transcendental equation \\
\bottomrule
\end{tabular}
\end{center}

\section{Applications in EDC}
\label{sec:bvp_applications}

\subsection{W Boson Mass}

The W mass comes from the lowest eigenvalue of the brane-localized
gauge field:
\begin{equation}
M_W^2 = \frac{\hbar^2 c^2}{\delta^2} \cdot \lambda_0 \quad \tagDc
\end{equation}

\begin{mechanism}{$M_W$ from BVP}
\textbf{Setup} \tagP{}:
\begin{itemize}[nosep]
\item W boson: First excited mode of brane gauge field
\item Boundary: Robin BC with $\kappa \sim 1/\delta$
\item Potential: Zero inside brane (simplest model)
\end{itemize}

\textbf{Result}:
The ground state of Eq.~(\ref{eq:bvp_generic}) gives:
\begin{equation}
\lambda_0 \approx \frac{\pi^2}{4} + O(\kappa\delta) \quad \tagDcApprox
\end{equation}

\textbf{Mass prediction}:
\begin{equation}
M_W \approx \frac{\pi \hbar c}{2\delta} \sim 80 \text{ GeV} \quad \tagDcApprox
\end{equation}
(with $\delta \sim 2.5 \times 10^{-3}$ fm)
\end{mechanism}

\subsection{Fermion Mass Tower}

The three lepton generations correspond to the first three eigenvalues:

\begin{center}
\begin{tabular}{ccc}
\toprule
\textbf{Mode} & \textbf{Generation} & \textbf{Particle} \\
\midrule
$n = 0$ & 1st & $e$ \\
$n = 1$ & 2nd & $\mu$ \\
$n = 2$ & 3rd & $\tau$ \\
\bottomrule
\end{tabular}
\end{center}

\subsection{Neutrino Suppression}

Neutrinos are edge modes (Chapter~\ref{ch:neutrinos}), giving exponentially
suppressed eigenvalues:
\begin{equation}
\lambda_\nu \sim e^{-2\kappa\delta} \quad \implies \quad
m_\nu \sim m_e \cdot e^{-\kappa\delta} \quad \tagDc
\end{equation}

\section{Numerical Methods}
\label{sec:numerical}
% ════════════════════════════════════════════════════════════

\subsection{Standard Approach}

For arbitrary $V(y)$, Eq.~(\ref{eq:bvp_generic}) must be solved numerically.

\textbf{Methods}:
\begin{enumerate}[nosep]
\item \textbf{Shooting method}: Integrate from boundaries, match in middle
\item \textbf{Finite differences}: Discretize $y$, solve matrix eigenvalue problem
\item \textbf{Spectral methods}: Expand in basis functions, solve algebraically
\end{enumerate}

\subsection{Numerical Standards}

As specified in Appendix~\ref{app:numerical_standards}:

\begin{enumerate}[nosep]
\item \textbf{Grid refinement}: Richardson extrapolation to verify convergence
\item \textbf{Cross-check}: Compare shooting vs finite difference
\item \textbf{Asymptotic}: Verify analytic limits ($\kappa \to 0, \infty$)
\end{enumerate}

\textbf{Tolerance}: $10^{-10}$ relative error for eigenvalues.

\subsection{Reference Implementation}

\begin{verbatim}
# Python pseudocode for Robin BC eigenvalue
def robin_eigenvalue(delta, kappa_p, kappa_m, n=0):
    """
    Solve transcendental equation for n-th eigenvalue
    with Robin BC: psi' + kappa*psi = 0
    """
    from scipy.optimize import brentq

    def residual(k):
        lhs = np.tan(k * delta)
        rhs = k * (kappa_p + kappa_m) / (k**2 - kappa_p*kappa_m)
        return lhs - rhs

    # Search in appropriate interval
    k_low = n * np.pi / delta + 0.01
    k_high = (n + 1) * np.pi / delta - 0.01

    k_n = brentq(residual, k_low, k_high)
    return k_n**2  # Return eigenvalue lambda
\end{verbatim}

\section{Convergence Analysis}
\label{sec:convergence}
% ════════════════════════════════════════════════════════════

\subsection{Grid Refinement}

For finite-difference solutions, we verify:
\begin{equation}
\lambda_N \to \lambda_{\text{exact}} + O(h^p) \quad \text{as } h \to 0
\end{equation}

where $h = \delta/N$ is the grid spacing and $p$ is the order of the method.

\textbf{Standard check}: Halve $h$, verify error reduces by factor $2^p$.

\subsection{Stability Analysis}

Eigenvalues are stable if:
\begin{equation}
\left|\frac{\partial \lambda}{\partial \kappa}\right| < \infty
\quad \text{and} \quad
\left|\frac{\partial \lambda}{\partial \delta}\right| < \infty
\end{equation}

\textbf{Observed}: All physical eigenvalues show smooth parameter dependence.
