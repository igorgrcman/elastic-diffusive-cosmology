\section*{Notation \& Conventions}

\subsection*{Units}
Natural units: $\hbar = c = 1$

Energy units: MeV (unless otherwise stated)

Length units: fm (unless otherwise stated)

\subsection*{5D vs 3D Quantities}
\begin{itemize}
\item 5D geometric quantities: no subscript (e.g., $\sigma$, $\delta$, $L_0$)
\item 3D/4D observables: subscript ``obs'' or explicit label (e.g., $m_p$, $G_F$)
\item Reduction factor: $\Cred$ (carries 5D$\to$3D normalization)
\end{itemize}

\subsection*{Epistemic Tags}
\begin{itemize}
\item \tagM{}: Pure mathematics
\item \tagP{}: Postulate (hypothesis)
\item \tagDer{}: Derived within model
\item \tagDc{}: Derived conditionally (requires dictionary)
\item \tagI{}: Identified (pattern match)
\item \tagCal{}: Calibrated (fitted)
\item \tagBL{}: Baseline (measurement)
\item \tagOpen{}: Open problem
\end{itemize}
