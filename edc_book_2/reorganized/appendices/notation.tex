\section{Units}
% ════════════════════════════════════════════════════════════

\textbf{Natural units}: $\hbar = c = 1$ unless otherwise stated.

\textbf{Practical conversions}:
\begin{align}
\hbar c &= 197.327 \text{ MeV} \cdot \text{fm} \\
(\hbar c)^2 &= 0.3894 \text{ GeV}^2 \cdot \text{mb}
\end{align}

\textbf{Energy units}: MeV for particle masses and nuclear scales;
GeV for electroweak scales.

\textbf{Length units}: fm (femtometer = $10^{-15}$ m) for nuclear/particle scales.

\textbf{Time units}: Seconds for lifetimes; $\hbar$/MeV $\approx 6.58 \times 10^{-22}$ s
for conversion.

\section{5D vs 3D/4D Quantities}
% ════════════════════════════════════════════════════════════

\textbf{Convention}: 5D geometric quantities carry no subscript;
3D/4D observables are explicitly labeled.

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{5D Quantity} & \textbf{3D/4D Observable} & \textbf{Relation} \\
\midrule
$\sigma$ (membrane tension) & $m_e$ (electron mass) & $m_e \propto \sigma^{1/3}$ \\
$\delta$ (brane thickness) & $M_W$ (W mass) & $M_W \sim \hbar c/\delta$ \\
$g_5$ (5D coupling) & $G_F$ (Fermi constant) & $G_F \propto g_5^2/M_W^2$ \\
$L_0$ (junction extent) & $r_p$ (proton radius) & $L_0 \sim r_p$ \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Reduction factor}: $C_{\text{red}}^{(X)}$ denotes the normalization
factor connecting 5D quantity to observable $X$.

\section{Transverse Coordinate Convention}
\label{sec:transverse_coord}
% ════════════════════════════════════════════════════════════

Throughout Part~II we use $\xi$ as the \emph{canonical transverse coordinate}
(normal to the brane). In some boundary-value-problem (BVP) sections we also
use $y$ as a local BVP coordinate. Unless explicitly stated otherwise, we
identify them by a trivial relabeling:
\begin{equation}
  \boxed{y \equiv \xi}
\end{equation}
with the domain written in either of the equivalent forms
\begin{equation}
  \xi \in \left[-\frac{\delta}{2},\frac{\delta}{2}\right]
  \quad \Longleftrightarrow \quad
  y \in \left[-\frac{\delta}{2},\frac{\delta}{2}\right].
\end{equation}

\textbf{Two distinct intervals appear}:
\begin{itemize}[nosep]
\item \textbf{Local wall region} $[-\delta/2, +\delta/2]$: The brane thickness
      region used in BVP analysis (Chapter~\ref{ch:bvp}).
\item \textbf{Compact bulk interval} $[0, \ell]$: The full 5th dimension used
      in KK reduction (OPR-19, Chapter~\ref{ch:foundation_params}).
\end{itemize}
These are \emph{different physical regions}; $\delta$ and $\ell$ are not
identified unless explicitly stated (see Scale Taxonomy,
Section~\ref{sec:scale_taxonomy}).

\toprule
\textbf{Tag} & \textbf{Name} & \textbf{Meaning} \\
\midrule
\tagM & Mathematics & Pure mathematical statement, provable within formal system \\

\begin{itemize}[nosep]

\end{itemize}

\section{Symbol Disambiguation}
% ════════════════════════════════════════════════════════════

The symbol $\kappa$ appears with two distinct meanings:

\begin{center}
\begin{tabular}{clcc}
\toprule
\textbf{Symbol} & \textbf{Meaning} & \textbf{Dimension} & \textbf{Chapters} \\
\midrule
$\kappa$ & Localization width (wavefunction decay) & length & \ref{ch:neutrinos}, \ref{ch:ckm} \\
$\kappa_T$ & Fermion--stress coupling & 1/mass$^3$ & \ref{ch:va_structure} \\
\bottomrule
\end{tabular}
\end{center}

Context disambiguates: $\kappa$ in exponential profiles $\exp(-|\xi|/\kappa)$ is the
localization width; $\kappa_T$ in $m(\xi) \sim \kappa_T T^{\xi\xi}$ is the stress coupling.

\subsection*{Volume Formulas}

\begin{align}
\text{Vol}(B^3) &= \frac{4\pi}{3}r^3 \quad \text{(3-ball in $\mathbb{R}^3$)} \\
\text{Vol}(S^3) &= 2\pi^2 r^3 \quad \text{(3-sphere in $\mathbb{R}^4$)}
\end{align}

\textbf{Warning}: Do not confuse $B^3$ (electron topology) with $S^3$ (proton arm topology).
