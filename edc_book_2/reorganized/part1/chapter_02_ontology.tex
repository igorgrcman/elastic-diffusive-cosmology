\section{Introduction: What Are Particles?}
\label{sec:ontology_intro}
% ════════════════════════════════════════════════════════════

In the Standard Model, particles are:
\begin{itemize}[nosep]
\item Point-like excitations of quantum fields
\item Classified by irreducible representations of the Poincaré group
\item Given masses via Yukawa couplings to the Higgs (parametrized, not explained)
\end{itemize}

\textbf{In EDC, particles are something completely different} \tagP{}:
\begin{itemize}[nosep]
\item \textbf{Topological defects} in a 5D membrane
\item Characterized by \textbf{winding numbers} and \textbf{junction configurations}
\item Masses emerge from \textbf{geometric properties} (volume, surface, tension)
\end{itemize}

\begin{statusbox}{The EDC Ontological Claim}
Particles are NOT point-like. They are extended 5D structures whose
3D projections we observe. The ``point-like'' behavior emerges from
the frozen projection mechanism (Chapter~\ref{ch:frozen}).
\end{statusbox}

\subsection{The Electron: Simplest Topological Defect}

\begin{mechanism}{Electron as B$^3$ Vortex}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Topology: $B^3$ (3-ball)---a simple spherical vortex in 5D
\item Configuration: $\text{Vol}(B^3) = \frac{4\pi}{3}$
\item Winding number: $W = -1$
\item Charge: Electric charge follows from winding orientation
\end{itemize}

\textbf{Stability} \tagM{}/\tagI{}:
The isoperimetric theorem \tagM{} guarantees the sphere is the \textit{unique}
configuration minimizing surface area for a given volume. Identifying this
with electron stability \tagI{} explains why no topological unwinding is possible.

\textbf{Mass origin}:
\begin{equation}
m_e \propto \sigma \cdot \text{Vol}(B^3)^{2/3} \quad \tagDcSym
\end{equation}
where $\sigma$ is the membrane tension.
\end{mechanism}

\subsection{The Proton: Y-Junction Structure}

From Book 1, the proton is established as a \textbf{Y-junction}---three membrane
arms meeting at a point:

\begin{mechanism}{Proton as Y-Junction}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Topology: Three arms meeting at $120^\circ$ angles (Steiner minimum)
\item Configuration: $(S^3)^3$ per arm, total: $(2\pi^2)^3$
\item Winding: Each arm carries $W = +1/3$ (quarks); total $W = +1$
\item Color: 3 arms $\to$ 3 QCD colors \tagI{}; 8 junction modes $\to$ 8 gluons \tagI{}
\end{itemize}

\textbf{Stability} \tagM{}/\tagI{}:
The Steiner/Lami theorem \tagM{} proves $120^\circ$ is the \textit{unique} angle
minimizing total arm length for fixed endpoints. Identifying this
with proton stability \tagI{} places the proton at a \textbf{local minimum} of the geometric action.

\textbf{Mass origin} (split derivation/dictionary):

\textit{Model result} \tagDerSym{}:
\begin{equation}
\frac{E_p}{E_e} = \frac{(2\pi^2)^3}{\frac{4\pi}{3}} = 6\pi^5 = 1836.118\ldots
\end{equation}

\textit{Dictionary step} \tagDc{}:
\begin{equation}
\frac{m_p}{m_e} \equiv \frac{E_p}{E_e} \quad\text{(mass = rest energy)}
\end{equation}
\end{mechanism}

\subsection{The Neutron: Excited Y-Junction}

The neutron is NOT a different particle---it's the \textit{same} Y-junction
in an excited configuration:

\begin{mechanism}{Neutron as Excited Junction}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Same topology as proton (Y-junction)
\item Different configuration: arms NOT at $120^\circ$ (off-Steiner)
\item Junction angle parameter: $\theta = 60^\circ$ (half-Steiner)
\item Metastable: Can relax toward $120^\circ$ (proton) + radiation
\end{itemize}

\textbf{Parametrization}:
\begin{equation}
\theta_n = (1 - Q) \times 60^\circ \quad \tagP
\end{equation}
For neutron: $Q = 0$, so $\theta = 60^\circ$.

\textbf{Decay}: Relaxation $\theta: 60^\circ \to 0^\circ$ releases energy into brane modes,
observed as $n \to p + e^- + \bar{\nu}_e$.
\end{mechanism}

\subsection{Quarks: Fractional Junction Arms}

Individual quarks are NOT free particles---they are the \textbf{arms} of a junction:

\begin{itemize}[nosep]
\item Each arm of the proton Y-junction is identified with a ``quark'' \tagI{}
\item Fractional charge $Q = +2/3$ or $-1/3$ from arm winding \tagDer{}
\item \textbf{Confinement} \tagDer{}: Infinite string energy prevents isolation (within model)
\item Color = arm label (3 arms $\to$ 3 colors) \tagI{}
\end{itemize}

The SU(3) color algebra structure emerges from the junction symmetry group \tagDer{};
the identification with QCD color is a dictionary step \tagI{}.

Bosons have a fundamentally different ontology---they are NOT topological defects

\subsection{Photon: Bulk Oscillation}

\begin{mechanism}{Photon as Bulk Mode}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Propagating wave in the 5D bulk
\item No localization to brane
\item Massless: No geometric confinement
\end{itemize}

\textbf{Why massless?}
A bulk wave has no characteristic scale from brane geometry.
\end{mechanism}

\subsection{W and Z Bosons: Brane-Localized Modes}

\begin{mechanism}{W/Z as Thick-Brane Modes}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Oscillatory modes \textit{localized} to the brane thickness $\delta$
\item Characteristic scale: $\lambda \sim \delta$
\item Mass from localization: $M_{W,Z} \sim \hbar c / \delta$
\end{itemize}

\textbf{Mass hierarchy}:
\begin{equation}
M_W \sim \frac{\hbar c}{\delta} \approx 80 \text{ GeV} \quad \tagI
\end{equation}
This requires $\delta \sim 2.5 \times 10^{-18}$ m (sub-proton scale).

\textbf{Full derivation}: Chapter~\ref{ch:electroweak}
\end{mechanism}

\subsection{Gluons: Junction Modes}

\begin{mechanism}{Gluons as Junction Oscillations}
\textbf{5D Structure} \tagP{}:
\begin{itemize}[nosep]
\item Oscillatory modes of the Y-junction itself
\item 8 modes from junction symmetry group \tagDer{}
\item Confined: Cannot propagate away from junction

\textbf{SU(3) identification} \tagI{}:
The 8 junction modes transform as the adjoint representation of SU(3)---this
is identified with the gluon octet. The group structure emerges from geometry,
but the identification ``junction modes = gluons'' is a dictionary step.
\end{mechanism}

\section{The Hopf Bridge: S$^3$ to S$^2$}
\label{sec:hopf_bridge}
% ════════════════════════════════════════════════════════════

A crucial conceptual issue: how do 5D objects appear 3D?

\begin{statusbox}{The S$^3$ vs S$^2$ Problem}
\textbf{Internal structure}: Particles have $S^3$ topology (3-sphere in 4D)

\textbf{Observed behavior}: We see $S^2$ angular momentum (spin, 3D rotation)

\textbf{Resolution}: The Hopf fibration $S^3 \to S^2$ with fiber $S^1$
\end{statusbox}

The \textbf{Hopf fibration} provides the bridge \tagM{}:
\begin{equation}
S^1 \longrightarrow S^3 \longrightarrow S^2
\end{equation}

\begin{itemize}[nosep]
\item Internal phase $\psi \in S^3$ describes full 5D orientation
\item Observed direction $\hat{t} \in S^2$ is what we measure
\item The $S^1$ fiber is the internal phase (related to U(1) gauge freedom)
\end{itemize}

\textbf{Physical consequence}: The ``hidden'' $S^1$ fiber is why fermions
require $720^\circ$ rotation to return to identity---they carry the full $S^3$
structure, not just $S^2$.

\begin{warningbox}{Critical: 3D Intuition Fails in 5D}
The most common errors in EDC derivations come from applying 3D formulas
to 5D objects. \textbf{MEMORIZE THESE}:

\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Quantity} & \textbf{3D Formula} & \textbf{5D/S$^3$ Formula} \\
\midrule
Volume & $\frac{4\pi}{3}r^3$ (B$^3$) & $2\pi^2 r^3$ (S$^3$) \\
Surface & $4\pi r^2$ (S$^2$) & $2\pi^2 r^2$ (S$^3$ ``area'') \\
Radial measure & $4\pi r^2 dr$ & $2\pi^2 r^3 dr$ (4D) \\
\bottomrule
\end{tabular}
\end{center}

\textbf{Error cost}: Using wrong formula gives factor $\sim 4.7$ error.
\end{warningbox}

\textbf{Rule}: Always verify which space you're integrating over before
using any volume or surface formula.
