\section{Physical Picture: Overlap Suppression}

\begin{mechanism}{CKM Hierarchy from Generation Separation}
\textbf{Step 1: Generations are spatially separated.}
The three quark generations ($u/c/t$ and $d/s/b$) are localized at different
positions along the extra dimension $\xi$. Each generation has a wavefunction
profile $f_i(\xi)$ peaked at its localization center \tagP{}.

\textbf{Step 2: Flavor mixing = overlap integral.}
The CKM element $V_{ij}$ is proportional to the overlap between up-type profile
$f_i^{(u)}$ and down-type profile $f_j^{(d)}$:
\begin{equation}
    V_{ij} \propto \int f_i^{(u)}(\xi) f_j^{(d)}(\xi) \, d\xi
    \quad \tagDc{}
\end{equation}

\textbf{Step 3: Separation kills off-diagonal mixing.}
If generations are well-separated (distance $\Delta\xi \gg$ profile width $\kappa$),
off-diagonal overlaps are \textbf{exponentially suppressed}:
\begin{equation}
    |V_{ij}| \sim e^{-|\xi_i - \xi_j|/2\kappa}
\end{equation}

\textbf{Step 4: Wolfenstein hierarchy emerges.}
With a single parameter $\Delta\xi/2\kappa \approx 1.5$:
\begin{itemize}[nosep]
    \item Adjacent generations: $|V_{us}|, |V_{cd}| \sim e^{-1.5} \equiv \lambda \approx 0.22$
    \item Skip one: $|V_{cb}|, |V_{ts}| \sim \lambda^2 \approx 0.05$
    \item Corner: $|V_{ub}|, |V_{td}| \sim \lambda^3 \approx 0.01$
\end{itemize}

\textbf{Step 5: Quarks vs leptons = localization strength.}
Quarks (color-charged, confined) have small $\kappa_q \Rightarrow$ near-diagonal CKM.
Leptons (color-neutral, edge modes) have large $\kappa_\ell \Rightarrow$ large PMNS angles.
\end{mechanism}

\subsection{Localized Profile Ansatz}

Each quark generation has a localized wavefunction profile \tagP{}:
\begin{align}
    f_i^{(u)}(\xi) &= N_u \exp\bigl(-|\xi - \xi_i^{(u)}|/\kappa\bigr) \\
    f_j^{(d)}(\xi) &= N_d \exp\bigl(-|\xi - \xi_j^{(d)}|/\kappa\bigr)
\end{align}

The CKM element is the overlap integral \tagDc{}:
\begin{equation}
    O_{ij} = \int d\xi\, f_i^{(u)}(\xi)\, f_j^{(d)}(\xi) \propto
    \exp\Bigl(-\frac{|\xi_i^{(u)} - \xi_j^{(d)}|}{2\kappa}\Bigr)
\end{equation}

\subsection{The Wolfenstein Mechanism}

Define inter-generation separation $\Delta\xi$. Then \tagDc{}:
\begin{align}
    |V_{ii}| &\sim 1 \quad \text{(same generation: maximal overlap)} \\
    |V_{i,i\pm 1}| &\sim e^{-\Delta\xi/2\kappa} \equiv \lambda \\
    |V_{i,i\pm 2}| &\sim e^{-2\Delta\xi/2\kappa} = \lambda^2 \\
    |V_{13}|, |V_{31}| &\sim \lambda^3 \quad \text{(corner elements)}
\end{align}

\textbf{Calibration:} To match Cabibbo angle $\lambda \approx 0.225$:
\begin{equation}
    \frac{\Delta\xi}{2\kappa} = -\ln\lambda \approx 1.49
    \quad \tagI{}
\end{equation}

This is \textbf{not a fit}---a single parameter determines the entire hierarchy.

\subsection{Phase Cancellation Theorem}

\begin{tcolorbox}[colback=yellow!5!white, colframe=yellow!50!black,
    title=\textbf{Theorem: Pure $\mathbb{Z}_3$ Gives $J = 0$}]
Any $\mathbb{Z}_3$ charge assignment with phases $\omega^{q_i}$ produces
a CKM matrix with \textbf{zero Jarlskog invariant} $J = 0$ \tagDc{}.

\textbf{Proof sketch:} The phases either cancel in the product
$\text{Im}(V_{us}V_{cb}V_{ub}^*V_{cs}^*)$ or are constrained by unitarity
to give real products.

\textbf{Consequence:} CP violation \textbf{cannot} arise from pure $\mathbb{Z}_3$.
\end{tcolorbox}

The 5° discrepancy remains \tagOpen{}.
