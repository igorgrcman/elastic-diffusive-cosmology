\subsection{Raw Sum Structure}
\label{subsec:raw-sum}

\subsection{Euler's Sum}
\label{subsec:euler-sum}

The fundamental result (Euler, 1734):
\begin{equation}
\label{eq:zeta-2}
\boxed{\sum_{n=1}^{\infty} \frac{1}{n^2} = \zeta(2) = \frac{\pi^2}{6}}
\end{equation}

\textbf{Derivation:} This is the Riemann zeta function at $s=2$, first computed
by Euler via the product formula for $\sin(\pi x)$.

\subsection{Conditions for Convergence}
\label{subsec:convergence-conditions}

The sum $\sum n^{-2}$ converges absolutely because:
\begin{equation}
\label{eq:convergence-test}
\sum_{n=1}^{\infty} \frac{1}{n^2} < 1 + \int_1^{\infty} \frac{dx}{x^2} = 1 + 1 = 2
\end{equation}

\textbf{Condition:} Flat overlap integrals (uniform coupling).

If overlap integrals $I_n \sim n^{-\alpha}$ for large $n$:
\begin{equation}
\label{eq:suppressed-sum}
\sum_{n=1}^{\infty} \frac{I_n^2}{n^2} \sim \sum_{n=1}^{\infty} \frac{1}{n^{2+2\alpha}} = \zeta(2+2\alpha)
\end{equation}

For $\alpha > 0$, convergence is faster.

\subsection{Zeta Function Regulator}
\label{subsec:zeta-regulator}

Regularize via analytic continuation:
\begin{equation}
\label{eq:zeta-reg}
\sum_{n=1}^{\infty} n^{-2-\epsilon} \bigg|_{\epsilon \to 0} = \zeta(2+\epsilon) = \zeta(2) + \epsilon \zeta'(2) + O(\epsilon^2)
\end{equation}

The finite part:
\begin{equation}
\label{eq:zeta-finite}
\lim_{\epsilon \to 0} \zeta(2+\epsilon) = \zeta(2) = \frac{\pi^2}{6}
\end{equation}

\subsection{Exponential/Heat Kernel Regulator}
\label{subsec:heat-kernel}

Alternative regularization:
\begin{equation}
\label{eq:heat-kernel-reg}
\sum_{n=1}^{\infty} \frac{e^{-\epsilon m_n^2}}{m_n^2} = \sum_{n=1}^{\infty} \frac{e^{-\epsilon n^2 \pi^2/L^2}}{n^2 \pi^2/L^2}
= \frac{L^2}{\pi^2} \sum_{n=1}^{\infty} \frac{e^{-\epsilon n^2 \pi^2/L^2}}{n^2}
\end{equation}

In the limit $\epsilon \to 0^+$:
\begin{equation}
\label{eq:heat-kernel-limit}
\lim_{\epsilon \to 0^+} \sum_{n=1}^{\infty} \frac{e^{-\epsilon n^2 \pi^2/L^2}}{n^2} = \sum_{n=1}^{\infty} \frac{1}{n^2} = \frac{\pi^2}{6}
\end{equation}

\subsection{Pauli-Villars Regulator}
\label{subsec:pv-regulator}

With Pauli-Villars mass $\Lambda$:
\begin{equation}
\label{eq:pv-reg}
\sum_{n=1}^{\infty} \frac{1}{m_n^2} - \sum_{n=1}^{\infty} \frac{1}{m_n^2 + \Lambda^2}
\end{equation}

As $\Lambda \to \infty$:
\begin{equation}
\label{eq:pv-limit}
\to \sum_{n=1}^{\infty} \frac{1}{m_n^2} = \frac{L^2}{\pi^2} \zeta(2)
\end{equation}

\section{Sum Convergence Proofs}
\label{app:sum-proofs}

\subsection{Comparison Test}
\label{app:comparison}

For the series $\sum_{n=1}^\infty 1/n^2$:
\begin{equation}
\label{eq:comparison-bound}
\sum_{n=1}^N \frac{1}{n^2} < 1 + \sum_{n=2}^N \frac{1}{n(n-1)} = 1 + \sum_{n=2}^N \left(\frac{1}{n-1} - \frac{1}{n}\right)
\end{equation}

Telescoping:
\begin{equation}
\label{eq:telescope}
= 1 + \left(1 - \frac{1}{N}\right) = 2 - \frac{1}{N} < 2
\end{equation}

\subsection{Integral Test}
\label{app:integral-test}

For $f(x) = 1/x^2$:
\begin{equation}
\label{eq:integral-bound}
\sum_{n=2}^\infty \frac{1}{n^2} < \int_1^\infty \frac{dx}{x^2} = 1
\end{equation}

Therefore:
\begin{equation}
\label{eq:sum-bound}
\sum_{n=1}^\infty \frac{1}{n^2} < 1 + 1 = 2
\end{equation}

\subsection{Exact Value via Fourier}
\label{app:fourier}

Consider $f(x) = x^2$ on $[-\pi, \pi]$:
\begin{equation}
\label{eq:fourier-f}
f(x) = \frac{\pi^2}{3} + 4\sum_{n=1}^\infty \frac{(-1)^n}{n^2} \cos(nx)
\end{equation}

At $x = \pi$:
\begin{equation}
\label{eq:fourier-pi}
\pi^2 = \frac{\pi^2}{3} + 4\sum_{n=1}^\infty \frac{1}{n^2}
\end{equation}

Solving:
\begin{equation}
\label{eq:zeta-2-fourier}
\sum_{n=1}^\infty \frac{1}{n^2} = \frac{\pi^2 - \pi^2/3}{4} = \frac{2\pi^2/3}{4} = \frac{\pi^2}{6}
\end{equation}

\subsection{Origin of Factor 1/8}

\subsection{Origin of $\pi^2/6$}
\label{app:factor-pi26}

\zeta(s) = \sum_{n=1}^\infty \frac{1}{n^s}
\end{equation}

At $s = 2$:
\begin{equation}
\label{eq:zeta-2-value}
\zeta(2) = \frac{\pi^2}{6}
\end{equation}

This is Euler's solution to the Basel problem (1734).

Alternative derivation via Parseval's theorem:
\begin{equation}
\label{eq:parseval}
\sum_{n=1}^\infty |c_n|^2 = \frac{1}{2\pi} \int_{-\pi}^{\pi} |f(x)|^2 dx

Thus:
\begin{equation}
\label{eq:zeta-from-parseval}

\subsection{Origin of $n\pi/L$}

The $S^1/\mathbb{Z}_2$ orbifold has fixed points at $y = 0$ and $y = L$.

For Neumann boundary conditions:

\partial_y \phi|_{y=0} = \partial_y \phi|_{y=L} = 0
\end{equation}

The mode solutions:

\label{eq:mode-solutions}

The eigenvalue equation:
\begin{equation}

\section{Regulator Independence Proofs}
\label{app:regulator-proofs}

\subsection{Zeta Regulator}
\label{app:zeta-proof}

Define:
\begin{equation}
\label{eq:zeta-s}
S(\epsilon) = \sum_{n=1}^\infty n^{-2-\epsilon}
\end{equation}

This is analytic for $\text{Re}(\epsilon) > -1$.

Laurent expansion around $\epsilon = 0$:
\begin{equation}
\label{eq:zeta-laurent}
S(\epsilon) = \zeta(2) + \epsilon \zeta'(2) + O(\epsilon^2)
\end{equation}

Taking $\epsilon \to 0$:
\begin{equation}
\label{eq:zeta-limit}
\lim_{\epsilon \to 0} S(\epsilon) = \zeta(2) = \frac{\pi^2}{6}
\end{equation}

\subsection{Heat Kernel Regulator}
\label{app:heat-proof}

Define:
\begin{equation}
\label{eq:heat-s}
S_t = \sum_{n=1}^\infty \frac{e^{-t n^2}}{n^2}
\end{equation}

As $t \to 0^+$:
\begin{equation}
\label{eq:heat-asymptotic}
S_t \sim \zeta(2) - \sqrt{\pi t} \zeta(1/2) + O(t)
\end{equation}

The $\zeta(1/2)$ term cancels in properly regulated expressions:
\begin{equation}
\label{eq:heat-regulated}
\lim_{t \to 0^+} \left(S_t + \sqrt{\pi t} \zeta(1/2)\right) = \zeta(2) = \frac{\pi^2}{6}
\end{equation}

\subsection{Agreement of Regulators}
\label{app:regulator-agree}

Both regulators give:
\begin{equation}
\label{eq:regulators-agree}
\sum_{n=1}^\infty \frac{1}{n^2} = \frac{\pi^2}{6}
\end{equation}

This is expected since the sum is \textbf{convergent} --- regularization only affects divergent sums.
