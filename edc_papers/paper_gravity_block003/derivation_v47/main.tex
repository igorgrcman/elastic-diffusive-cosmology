\label{subsec:standard-norm}

For $SU(N)$ in the fundamental representation, the canonical normalization is:

\begin{equation}
\label{eq:trace-derivation}
\mathrm{Tr}(T^a T^b) = T(R) \delta^{ab}
\end{equation}
where $T(R)$ is the Dynkin index. For the fundamental of $SU(N)$:
\begin{equation}
\label{eq:dynkin-fund}
T(\square) = \frac{1}{2}
\end{equation}

\subsection{$SU(2)$ Generators}
\label{subsec:su2-generators}

For $SU(2)_{L,R}$, the generators in the fundamental (doublet) are:
\begin{equation}
\label{eq:su2-generators}
T^a = \frac{1}{2} \sigma^a, \quad a = 1,2,3
\end{equation}
where $\sigma^a$ are Pauli matrices.

Trace verification:
\begin{equation}
\label{eq:su2-trace-verify}
\mathrm{Tr}(T^a T^b) = \frac{1}{4} \mathrm{Tr}(\sigma^a \sigma^b) = \frac{1}{4} \cdot 2 \delta^{ab} = \frac{1}{2} \delta^{ab} \quad \checkmark
\end{equation}

For $T_{3R}$ specifically:
\begin{equation}
\label{eq:t3r-explicit}
T_{3R} = \frac{1}{2} \sigma^3 = \frac{1}{2} \begin{pmatrix} 1 & 0 \\ 0 & -1 \end{pmatrix}
\end{equation}

Trace:
\begin{equation}
\label{eq:t3r-trace}
\mathrm{Tr}(T_{3R}^2) = \mathrm{Tr}\left(\frac{1}{4} \begin{pmatrix} 1 & 0 \\ 0 & 1 \end{pmatrix}\right) = \frac{1}{4} \cdot 2 = \frac{1}{2}
\end{equation}

\label{subsec:b-minus-l}

The $(B-L)$ generator comes from $SU(4)_C \to SU(3)_C \times U(1)_{B-L}$.

In the fundamental $\mathbf{4}$ of $SU(4)_C$:

\label{eq:b-minus-l-explicit}

The first three entries correspond to quarks (baryon number $B = 1/3$, lepton number $L = 0$),

\label{eq:b-minus-l-trace}

\subsection{Trace Factorization}
\label{app:trace-factor}

\begin{equation}
\label{eq:trace-product}
\mathrm{Tr}_V(A \otimes B) = \mathrm{Tr}_{V_1}(A) \cdot \mathrm{Tr}_{V_2}(B)
\end{equation}

Applied to $T_{3R}$ and $(B-L)$:
\begin{equation}
\label{eq:cross-trace}
\mathrm{Tr}(T_{3R} \cdot (B-L)) = \mathrm{Tr}_{SU(2)_R}(T_{3R}) \cdot \mathrm{Tr}_{SU(4)_C}(B-L)

\end{align}

Therefore:
\begin{equation}
\label{eq:cross-vanish}
\mathrm{Tr}(T_{3R} \cdot (B-L)) = 0 \times 0 = 0 \quad \checkmark
\end{equation}
