\begin{align}
\label{eq:burden-mech}

\label{eq:burden-params}
N_{\text{params}} &= \text{count of continuous free parameters} \\
\label{eq:burden-tuning}

\end{align}

Lower burden is preferred:
\begin{equation}
\label{eq:burden-ordering}

\subsection{Lexicographic Decision Rule}
\label{subsec:lexicographic}

The complete decision uses lexicographic ordering on scoring vectors.

\begin{definition}[Scoring Vector]
\label{def:scoring-vector}
Each track $T$ has scoring vector:
\begin{equation}
\label{eq:scoring-vector}
\vec{V}(T) = \left(\text{S0}, \text{S1}, S_{\text{vac}}, B, -H, \dim(G), \Delta_{\text{rank}}, n_{\text{exotic}}\right)
\end{equation}
where:
\begin{align}
\label{eq:sv-s0}
\text{S0} &= 0 \text{ if Stage 0 passes, else } 1 \\
\label{eq:sv-s1}
\text{S1} &= 0 \text{ (PASS)}, 1 \text{ (CONDITIONAL)}, 2 \text{ (FAIL)}
\end{align}
\end{definition}

\begin{theorem}[Selection Rule]
\label{thm:selection}
The selected track is:
\begin{equation}
\label{eq:selection-rule}
T^* = \arg\min_{T \in \mathcal{T}} \vec{V}(T)
\end{equation}
where the minimum uses lexicographic ordering. If multiple tracks achieve
the same minimum, return \texttt{UNRESOLVED}.
\end{theorem}

\section{Reviewer Traps}
\label{sec:traps}
% =============================================================================

The following common pitfalls are documented:

\begin{enumerate}[label=\textbf{Trap \arabic*}, leftmargin=*]

\item \textbf{Vacuum energy absolute value}:
The $\Delta E_{\text{vac}}$ score is a \emph{relative} measure, not an absolute energy.
Only differences between BC configurations are meaningful.

\item \textbf{Regulator dependence}:
The finite part of $\Delta E_{\text{vac}}$ is regulator-invariant (v37 verified zeta/heat-kernel
agreement). The divergent parts cancel in the difference.

\item \textbf{Weyl vs Dirac counting}:
All fermion counts use LH Weyl basis. A Dirac fermion = 2 Weyl fermions.
Multiplicities in SoT are Weyl counts.

\item \textbf{BC to zero-mode mistakes}:
Mixed BC (ND/DN) $\Rightarrow$ no zero-mode. This is not optional; it's
topologically enforced by the orbifold boundary conditions.

\item \textbf{Hidden tuning in brane mass}:
Brane mass terms require parameters not fixed by symmetry. This constitutes
``hidden tuning'' and triggers CONDITIONAL status.

\item \textbf{Hosotani VEV assumption}:
The Hosotani mechanism requires a specific VEV configuration. This is a
dynamical assumption, not a UV boundary condition.

\item \textbf{PASS vs anomaly-free confusion}:
All four tracks are anomaly-free. ``PASS'' refers to admissibility (exotic gating),
not anomaly status.

\item \textbf{Stage ordering matters}:
Earlier stages have priority. A track cannot compensate for Stage 1 failure
with excellent Stage 2 score.

\item \textbf{Lexicographic vs weighted sum}:
The scoring is \emph{lexicographic}, not a weighted sum. Component $i$ is only
compared if components $1, \ldots, i-1$ are tied.

\item \textbf{Hash collision probability}:
The 16-hex-character hash has $2^{64}$ possible values. Collision probability
for 4 tracks is negligible.

\item \textbf{Tie-breaker scope}:
Tie-breakers only apply if all stages produce ties. In practice, Stage 2
(vacuum energy) usually breaks ties.

\item \textbf{UNRESOLVED is valid}:
The selector may legitimately return UNRESOLVED if the SoT data is insufficient
to distinguish tracks. This is an explicit outcome, not an error.

\item \textbf{Mechanism dictionary completeness}:
All exotic decoupling mechanisms must be enumerated in the SoT. Missing
mechanisms cause FAIL, not PASS.

\item \textbf{Fraction arithmetic precision}:
All hypercharge calculations use exact rational arithmetic (Python \texttt{Fraction}).
No floating-point errors possible.

\item \textbf{Score sign convention}:
For $S_{\text{vac}}$ and $B$: \emph{lower is better}. For $H$: \emph{higher is better}.
The scoring vector uses $-H$ to maintain ``lower is better'' convention.

\item \textbf{Rank drop interpretation}:
$\Delta_{\text{rank}} = \text{rank}(G) - \text{rank}(H)$ measures how much structure
is broken. Higher rank drop means more symmetry breaking.

\item \textbf{Generator counting}:
$\dim(G)$ counts adjoint generators, not fundamental representation dimension.
$\dim(SU(N)) = N^2 - 1$.

\item \textbf{Hard gate vs soft filter}:
Stage 0 gates are \emph{hard}: any failure eliminates the track. Stages 1--4
are \emph{soft}: they rank but don't eliminate (unless all tracks fail).

\end{enumerate}

\section{Lexicographic Ordering Proof}
\label{app:lex-proof}

\begin{definition}[Lexicographic Order]
\label{def:lex-order}
For vectors $\vec{u} = (u_1, \ldots, u_n)$ and $\vec{v} = (v_1, \ldots, v_n)$:
\begin{equation}
\label{eq:lex-order-def}
\vec{u} <_{\text{lex}} \vec{v} \iff \exists k : u_k < v_k \text{ and } \forall i < k : u_i = v_i
\end{equation}
\end{definition}

\begin{proposition}[Selection Well-Definedness]
\label{prop:well-defined}
The lexicographic order on scoring vectors is a total order with unique minimum
(when tracks differ).
\end{proposition}

\begin{proof}
Scoring vectors have 8 numerical components. Since all components are computed
from discrete SoT data, equal vectors imply identical track characteristics.
The lexicographic order on finite sequences of comparable elements is a total
order, so either $\vec{V}(T_1) <_{\text{lex}} \vec{V}(T_2)$,
$\vec{V}(T_1) = \vec{V}(T_2)$, or $\vec{V}(T_1) >_{\text{lex}} \vec{V}(T_2)$.
\end{proof}
